When manipulating small multi-dimensional tensors, operations like tl.permute, tl.reshape, or tl.split force the tensor to round-trip through Shared Memory (Smem) via ttg.convert_layout -> st.shared / ld.shared.

This happens even when the tensor is small enough to fit entirely in registers, and the permutation is purely a logical re-indexing of the data, or requires at most simple warp-level register shuffles (shfl.sync). This unnecessary Smem traffic causes L1/TEX cache bottlenecks and bank conflicts.

There was an option to lift the restriction,

triton/lib/Analysis/Utility.cpp

Lines 1310 to 1314 in 96a22b5

if (to_vector(layout.getOutDimNames()) ==
SmallVector<StringAttr, 2>{kRegister, kLane}) {
auto factors = getWarpLayoutConvertDecomposition(srcTy, dstTy, 32);
return (factors.mixedTranspositions.size() < 2);
}
and change it to:

  if (to_vector(layout.getOutDimNames()) ==
      SmallVector<StringAttr, 2>{kRegister, kLane}) {
    return true;

but that change might have a broader effect, and I would guess that it will affect other kernels, where tensors are bigger. This is why Triton is checking if avoiding shared memory will pay off.

The PR is solving the problem that was reported in #10342

All existing lit tests passed, I have added new ones to check this functionality.
I added a simple reproducer of the problem to test_core.py.

import argparse
import pathlib
import triton
import torch
import sys

def generate_ttgir(tsize, dtype_str, c, order_dst, iters, target_str):
    tpw_src = [1, 32]
    spt_src = [tsize // tpw_src[0], tsize // tpw_src[1]]
    ord_src = [1, 0]
    
    tpw_dst = [c, 32 // c]
    spt_dst = [tsize // tpw_dst[0], tsize // tpw_dst[1]]
    ord_dst = order_dst
    
    return f"""
#blocked = #ttg.blocked<{{sizePerThread = [{spt_src[0]}, {spt_src[1]}], threadsPerWarp = [{tpw_src[0]}, {tpw_src[1]}], warpsPerCTA = [1, 1], order = [{ord_src[0]}, {ord_src[1]}]}}>
#test = #ttg.blocked<{{sizePerThread = [{spt_dst[0]}, {spt_dst[1]}], threadsPerWarp = [{tpw_dst[0]}, {tpw_dst[1]}], warpsPerCTA = [1, 1], order = [{ord_dst[0]}, {ord_dst[1]}]}}>
module attributes {{"ttg.num-ctas" = 1 : i32, "ttg.num-warps" = 1 : i32, ttg.target = "{target_str}", "ttg.threads-per-warp" = 32 : i32}} {{
  tt.func public @kernel(%arg0: !tt.ptr<{dtype_str}> {{tt.divisibility = 16 : i32}}, %arg1: !tt.ptr<{dtype_str}> {{tt.divisibility = 16 : i32}}) attributes {{noinline = false}} {{
    %c1_i32 = arith.constant 1 : i32
    %iters_i32 = arith.constant {iters} : i32
    %c0_i32 = arith.constant 0 : i32
    %0 = tt.make_range {{end = {tsize} : i32, start = 0 : i32}} : tensor<{tsize}xi32, #ttg.slice<{{dim = 0, parent = #blocked}}>>
    %1 = tt.expand_dims %0 {{axis = 0 : i32}} : tensor<{tsize}xi32, #ttg.slice<{{dim = 0, parent = #blocked}}>> -> tensor<1x{tsize}xi32, #blocked>
    %2 = tt.make_range {{end = {tsize} : i32, start = 0 : i32}} : tensor<{tsize}xi32, #ttg.slice<{{dim = 1, parent = #blocked}}>>
    %3 = tt.expand_dims %2 {{axis = 1 : i32}} : tensor<{tsize}xi32, #ttg.slice<{{dim = 1, parent = #blocked}}>> -> tensor<{tsize}x1xi32, #blocked>
    %4 = tt.broadcast %1 : tensor<1x{tsize}xi32, #blocked> -> tensor<{tsize}x{tsize}xi32, #blocked>
    %5 = tt.broadcast %3 : tensor<{tsize}x1xi32, #blocked> -> tensor<{tsize}x{tsize}xi32, #blocked>
    %6 = arith.addi %4, %5 : tensor<{tsize}x{tsize}xi32, #blocked>
    %7 = tt.splat %arg0 : !tt.ptr<{dtype_str}> -> tensor<{tsize}x{tsize}x!tt.ptr<{dtype_str}>, #blocked>
    %8 = tt.addptr %7, %6 : tensor<{tsize}x{tsize}x!tt.ptr<{dtype_str}>, #blocked>, tensor<{tsize}x{tsize}xi32, #blocked>
    %9 = tt.load %8 : tensor<{tsize}x{tsize}x!tt.ptr<{dtype_str}>, #blocked>

    %10 = scf.for %arg2 = %c0_i32 to %iters_i32 step %c1_i32 iter_args(%arg3 = %9) -> (tensor<{tsize}x{tsize}x{dtype_str}, #blocked>)  : i32 {{
      %x = ttg.convert_layout %arg3 : tensor<{tsize}x{tsize}x{dtype_str}, #blocked> -> tensor<{tsize}x{tsize}x{dtype_str}, #test>
      %res = arith.addi %x, %x : tensor<{tsize}x{tsize}x{dtype_str}, #test>
      %13 = ttg.convert_layout %res : tensor<{tsize}x{tsize}x{dtype_str}, #test> -> tensor<{tsize}x{tsize}x{dtype_str}, #blocked>
      scf.yield %13 : tensor<{tsize}x{tsize}x{dtype_str}, #blocked>
    }}
    %11 = tt.splat %arg1 : !tt.ptr<{dtype_str}> -> tensor<{tsize}x{tsize}x!tt.ptr<{dtype_str}>, #blocked>
    %12 = tt.addptr %11, %6 : tensor<{tsize}x{tsize}x!tt.ptr<{dtype_str}>, #blocked>, tensor<{tsize}x{tsize}xi32, #blocked>
    tt.store %12, %10 : tensor<{tsize}x{tsize}x!tt.ptr<{dtype_str}>, #blocked>
    tt.return
  }}
}}
"""

def main():
    parser = argparse.ArgumentParser(description="Run layout conversion benchmarks.")
    parser.add_argument("--tsize", type=int, default=32, help="Tensor size (tsize x tsize). Must be multiple of 32.")
    parser.add_argument("--iters", type=int, default=100000, help="Number of loop iterations in the kernel.")
    parser.add_argument("--order", type=int, nargs=2, default=[1, 0], help="Order of the destination layout (e.g. 1 0 or 0 1).")
    args = parser.parse_args()
    
    tsize = args.tsize
    iters = args.iters
    order_dst = args.order
    
    if tsize % 32 != 0:
        print("Error: --tsize must be a multiple of 32")
        sys.exit(1)
        
    dtypes = [
        ("i8", torch.int8),
        ("i16", torch.int16),
        ("i32", torch.int32),
        ("i64", torch.int64),
    ]
    
    c_values = [1, 2, 4, 8, 16, 32]
    
    cc = torch.cuda.get_device_capability()
    target_str = f"cuda:{cc[0]}{cc[1]}"
    
    print(f"Benchmarking with TSIZE={tsize}, Iterations={iters}, OrderDst={order_dst}")
    print("-" * 65)
    
    header = f"{'c (tPW_dst)':<15} |" + "".join([f" {d[0]:>8} |" for d in dtypes])
    print(header)
    print("-" * len(header))
    
    for c in c_values:
        row_str = f"c={c:<2} ([{c:2}, {32//c:2}])  |"
        for d_str, t_dtype in dtypes:
            ttgir = generate_ttgir(tsize, d_str, c, order_dst, iters, target_str)
            
            temp_file = pathlib.Path(f"test_tmp_{d_str}_{c}.ttgir")
            temp_file.write_text(ttgir)
            
            try:
                kernel = triton.compile(str(temp_file))
                a = torch.arange(tsize * tsize, device="cuda", dtype=t_dtype).reshape(tsize, tsize)
                b = torch.empty_like(a)
                ms, _, _ = triton.testing.do_bench(lambda: kernel[(1, 1, 1)](a, b), quantiles=[0.5, 0.2, 0.8])
                row_str += f" {ms:8.2f} |"
            except Exception as e:
                # print(f"Error compiling for {d_str} with c={c}: {e}")
                row_str += f" {'ERR':>8} |"
            
            if temp_file.exists():
                temp_file.unlink()
                
        print(row_str)

if __name__ == "__main__":
    main()
